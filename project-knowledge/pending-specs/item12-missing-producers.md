def regen_nfos_from_history(
    *,
    site_id: Optional[str] = None,
    overwrite: bool = False,
    max_files: int = 1000,
    dry_run: bool = False,
    download_dir: str = "",
) -> dict:
    """Generate (or update) NFO sidecars for every existing file in
    history. Uses whatever metadata is recorded -- TPDB if it was
    enriched, otherwise just filename + message.

    ``download_dir`` is what a recorded basename is RESOLVED against.
    runner_transport.py:990 binds ``filename = final_path.name`` -- the ONLY
    binding of that name in the module -- so ``history.filename`` is a bare
    BASENAME in production. The previous body tested ``Path(fn).exists()``,
    which resolves against the process CWD: every production row failed, was
    counted ``missing_files``, and NO sidecar was ever written. Same root
    cause cut 25b fixed for list_missing_from_disk and size_drift_scan; this
    function was left behind because its only test seeds an ABSOLUTE
    filename, so production's input form was outside that test's denominator.

    A row that cannot be PLACED is not missing. ``unresolved`` counts
    "ambiguous" (the basename matches several files -- writing a sidecar next
    to a guess is worse than writing none) and "unknown" (no download_dir),
    which _resolve_recorded deliberately keeps out of "absent".

    Returns {written, skipped, missing_files, unresolved, errors,
    considered, limit_hit, download_dir}."""
    out = {"written": 0, "skipped": 0, "missing_files": 0,
           "unresolved": 0, "errors": 0, "considered": 0,
           "limit_hit": False, "download_dir": download_dir}
    try:
        from . import db as _db
        sql = "SELECT * FROM history WHERE status='done' AND filename != ''"
        params: list = []
        if site_id:
            sql += " AND site_id = ?"
            params.append(site_id)
        sql += " ORDER BY id DESC LIMIT ?"
        params.append(int(max_files))
        with _db.db_conn() as cx:
            rows = [dict(r) for r in cx.execute(sql, params).fetchall()]
    except Exception:
        return out
    out["considered"] = len(rows)
    out["limit_hit"] = len(rows) >= int(max_files)
    index = _basename_index(download_dir)
    for r in rows:
        fn = r.get("filename") or ""
        path, state = _resolve_recorded(fn, download_dir, index)
        if state == "absent":
            out["missing_files"] += 1
            continue
        if state != "resolved" or path is None:
            out["unresolved"] += 1
            continue
        target = str(path)
        nfo_path = Path(target).with_suffix(".nfo")
        if nfo_path.exists() and not overwrite:
            out["skipped"] += 1
            continue
        meta = {
            "title": Path(target).stem,
            "sorttitle": Path(target).stem,
            "plot": r.get("message", ""),
            "studio": r.get("site_name", ""),
            "source_url": r.get("url", ""),
        }
        # v3.66.522 (VR-P06): honor dry_run (the handler defaults it True =
        # preview). Count the row as a would-write but never touch disk.
        if dry_run:
            out["written"] += 1
            continue
        if write_nfo(target, meta, overwrite=overwrite):
            out["written"] += 1
        else:
            out["errors"] += 1
    return out
