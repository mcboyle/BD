# Bulk Downloader v3.0 — runtime dependencies
# Install: pip install -r requirements.txt
# Then:    playwright install chromium

flask>=3.0,<4.0
# Playwright pinned <2.0 because the .spec hard-codes paths under
# playwright._impl.* — a 2.x major bump may reorganize those.
# MOD-6 (v3.66.678): floor raised to 1.61 to unify the service venv (1.61.0) with
# the sandbox prestage (was 1.60.0) — both satisfied the old >=1.45 range, hiding
# the cross-environment drift. dep_freshness now flags anything below the floor.
playwright>=1.61,<2.0
# httpx is used for direct HTTP downloads (faster, with progress and resume).
# If unavailable, the worker falls back to Playwright's save_as.
httpx>=0.25,<1.0

# Template discovery and deep-detection paths parse captured HTML with
# BeautifulSoup. These paths fail open when bs4 is absent, but that silently
# disables selector fitness scoring on a clean install.
beautifulsoup4>=4.12,<5.0

# v3.66.848: lxml backs the ARIA audit (bulk_downloader/accessibility.py:185)
# and XPath selector evaluation (selector_playground.py:56). Both fail open
# when it is absent -- the audit returns a single "Install lxml to run ARIA
# audit" finding and XPath selectors return an error string -- so a missing
# lxml degrades silently, exactly as the beautifulsoup4 note above describes.
# It was never declared here, which is why `pip check` reported clean while
# tests/test_v3_43_80_modules.py's TestAccessibility and TestSynthetic failed
# on a fresh container: pip cannot see a requirement that was never written
# down. diagnostics_bundle.py:130 still lists it among the optional deps it
# REPORTS on, which is correct -- reporting availability and declaring an
# install floor are different jobs.
# NOT pure Python: lxml is a C extension over libxml2. It ships manylinux,
# macOS and Windows wheels, so the common installs need no toolchain, but a
# platform without a wheel will need libxml2/libxslt headers to build.
lxml>=5.0,<7.0

# Phase 15.4: TLS fingerprint impersonation. Optional — falls back to httpx
# if not installed. Provides a drop-in `requests`-style API that mimics real
# Chrome's TLS handshake (JA3, ALPN, HTTP/2 SETTINGS) at the wire level —
# defeats fingerprint-based bot detection that classifies based on TLS
# alone, not just user-agent strings.
curl_cffi>=0.7,<1.0

# v3.43.14: Secure password storage.
#
# keyring        - OS keychain access (Windows Credential Manager,
#                  macOS Keychain, Linux Secret Service). The recommended
#                  storage for passwords on systems that have a keychain.
# cryptography   - AES-GCM for the master-password fallback backend (used
#                  when no keychain is available, e.g. headless Linux).
#
# Both have pure-wheel Windows installs. If they fail to install (rare),
# the app falls back to plaintext storage in sites_config.json — same
# behavior as v3.43.13 and earlier — but the migration UI won't appear.
keyring>=24.0,<26.0
cryptography>=42.0,<46.0

# v3.66.681 (B2/P6): OIDC / SSO login. authlib.jose verifies the provider's
# id_token (RS256) against its JWKS. Only needed when SSO is enabled;
# imported function-locally so the app runs without it otherwise.
authlib>=1.3,<2.0

# v3.43.64: MP4 metadata tagging via the mutagen library. Used by
# bulk_downloader/mp4_metadata.py to embed title/performer/album/date/
# comment/cover atoms into MP4 files after download. Pure Python, no
# compilation step, ~600KB installed. Fail-open: if missing, tagging
# is a silent no-op (the file downloads fine without tags).
mutagen>=1.47,<2.0

# v3.62.2: openpyxl powers the XLSX bulk-import template (a formatted
# downloadable workbook with a template-id dropdown) and parsing of
# uploaded .xlsx import files. Pure Python, no compilation step.
# bulk_downloader/csv_bulk.py: CSV import works without it; XLSX
# import/template degrade with a clear error if it is ever absent.
openpyxl>=3.1,<4.0

# v3.63.6: psutil powers the cpu_ram dashboard widget
# (bulk_downloader/app_widgets_api.py::_collect_data → cpu_pct/ram_pct/
# cpu_cores/ram_total_fmt). The widget code soft-imports psutil and
# fails open if it is missing — the response keys are then None, and
# the frontend renders "—% / —%" instead of misleading "0% / 0%".
# Pre-v3.63.6 psutil was documented as optional in doctor.py but not
# pinned anywhere, so fresh installs always missed it and the widget
# always displayed 0% / 0%. Stash diagnosis 2026-05-21 confirmed.
# Pure Python wheel; no compilation step on Linux or Windows.
psutil>=5.9,<7.0

# v3.66.794 (F0.4): production WSGI server. The werkzeug dev server that
# app.run() starts is single-process and crashes under concurrent load
# (Task-E fuzz recon). In production (debug off) downloader_ui serves under
# waitress -- a real threaded WSGI server. Soft/fail-open: imported only in
# _serve_wsgi, and if waitress is absent the app falls back to the werkzeug
# dev server rather than refusing to boot. Pure Python wheel, no build step.
# Size the pool with BD_WSGI_THREADS (default 30): each long-lived /api/stream
# SSE connection pins one worker thread for its life.
waitress>=3.0,<4.0

# Capture and validation must use the real pytest runner on every supported
# install, including production/test-box deployments that install only this
# core manifest.
pytest>=7.0,<9.0
pytest-xdist>=3.6,<4.0
