# Bulk Downloader — OPTIONAL runtime dependencies
# Install: pip install -r requirements-optional.txt
#
# Each of these is gracefully degraded — the rest of the app works fine
# without them. They're separated from requirements.txt so a build-tool
# failure on Windows (e.g. missing VC++ for curl_cffi) doesn't block the
# main install.

# Plex alternate backend. Enables the opt-in plexapi library-sync and richer
# Plex read operations; the built-in raw HTTP backend remains the fallback.
plexapi

# Subtitle sidecars. subliminal searches providers and writes subtitles;
# babelfish converts operator language codes for those searches.
subliminal
babelfish

# Desktop tray icon. Pillow supplies the image support from requirements.txt;
# pystray supplies the platform tray/menu integration.
pystray

# Terminal dashboard. Enables the live Rich TUI; plain stdout remains the
# fallback when this optional package is not installed.
rich

# Plugin key/value Postgres fallback. The default remains SQLite; installing
# this distribution provides the legacy ``psycopg2`` import used for an
# explicitly supplied Postgres DSN when the modern psycopg driver is absent.
psycopg2-binary

# Phase 15.4: TLS fingerprint impersonation. Provides a drop-in `requests`-
# style API that mimics real Chrome's TLS handshake (JA3, ALPN, HTTP/2
# SETTINGS) at the wire level — defeats fingerprint-based bot detection
# that classifies based on TLS alone, not just user-agent strings. App
# falls back to httpx if not installed.
curl_cffi>=0.7,<1.0

# Phase 16.43: Web push notifications. Both libraries needed together —
# cryptography for VAPID keypair gen, pywebpush for delivery. App hides
# the bell button when either is missing.
cryptography>=40,<50
pywebpush>=2.0,<3.0

# Phase 37: QR code generation for cross-device pairing. The /api/pair
# endpoint returns SVG QR codes when this is installed; without it,
# the pairing modal falls back to a copyable URL.
qrcode>=7.0,<9.0

# v3.43.56: Cloudflare-evasion library. More comprehensive bot-
# detection patcher than the built-in STEALTH_JS — adds evasions
# for navigator.permissions, WebGL vendor/renderer leak, chrome.app
# absence, the iframe.contentWindow null check, plus several others
# that Cloudflare's heavily-obfuscated detection script checks. Use
# by enabling `use_stealth_library` in the site config. App falls
# back to STEALTH_JS if not installed (no error).
playwright-stealth>=1.0,<3.0

# v3.43.63: library-extractor fast path. Maintained PyPI extractor
# libraries (EchterAlsFake family) for adult-content sites. Each library
# is independent and may be installed à la carte; bulk_downloader's
# `extractors` module gracefully skips any not installed. Use by
# enabling `use_library_extractor: true` per site, OR the relevant
# templates ship with this on by default in v3.43.63.
#
# The shared base API (eaf_base_api) is pulled in transitively by most
# of these but listed explicitly so a minimal install still works.
#
# v3.47.1: any extractor pinned `>=N.M` that's been pulled from PyPI
# is commented out below. Uncomment + adjust pin when a fork/replacement
# emerges. The runner gracefully degrades — sites without their
# library extractor fall back to the generic teach-based path.
#
# v3.63.5: the `xnxx_api` and `beeg_api` pins were both removed
# entirely (not just commented out). `xnxx_api` was pulled from PyPI
# and unfindable; `beeg_api` installs hit intermittent 404s on a wheel
# that wasn't being maintained. The adapters in `extractors.py` for
# both sites still exist and still call `_try_import("xnxx_api" /
# "beeg_api")`, which now correctly returns None on every install
# (since neither package is in any requirements file) and the
# adapter returns `library_not_installed`, BD falls through to the
# generic teach-based path. To restore either, add the line back
# here AND verify the package still exists on PyPI first — the pins
# were removed because the failures were noisy on every fresh
# install, not because the underlying adapter logic was broken.
eaf_base_api>=2.5
phub>=4.0
xvideos_api>=1.7
xhamster_api>=1.6
spankbang_api>=1.4
eporner_api>=1.5
hqporner_api>=1.5
youporn_api>=1.4
porntrex_api>=1.3
missav_api>=1.2
xfreehd_api>=1.2
porngo_api>=1.2

# v3.43.63: m3u8 manifest parser (optional — used by hls_downloader for
# pre-flight variant inspection. Without it, ffmpeg still handles the

m3u8>=3.6,<6

# v3.43.70: apprise notifications. One library, 80+ services
# (Telegram, Discord, Pushover, ntfy, Gotify, Slack, Pushbullet,

# Fail-open in BD — the notify_apprise module reports

# v3.43.72: perceptual video dedup. Computes a 64-bit pHash from
# evenly-spaced frames of a video; lets BD catch duplicate scenes
# across networks even when filenames + file sizes differ. Requires
# ffmpeg on PATH (which BD already needs for HLS/library extractors).
# Fail-open in BD — the dedup module reports `videohash_not_installed`
# and skips registration when missing.
videohash>=2.1,<3.0

# v3.43.73: Scrapling adaptive selectors + Turnstile bypass.
# Scrapling is an undetectable web-scraping library that adds two

# Turnstile bypass via StealthyFetcher. Both are opt-in per site;
# fail-open everywhere — missing library reports `not_available`
# and BD's existing teach path runs unchanged.
scrapling>=0.2,<1.0
